## How To Use These Components

### 1. Vector3 Math
```javascript
import Vector3 from "./utils/Vector3.js";

// Create vectors
const point1 = new Vector3(0, 64, 0);
const point2 = new Vector3(10, 80, 10);

// Calculate distance
const distance = Vector3.distance(point1, point2);

// Normalize direction
const direction = Vector3.subtract(point2, point1).normalized();

// Draw line with particles
const step = Vector3.multiply(direction, 1);
let current = point1;
for (let i = 0; i < 10; i++) {
    dimension.spawnParticle("minecraft:balloon_gas_particle", current.toObject());
    current = Vector3.add(current, step);
}
```

### 2. Storage Database
```javascript
import { Storage } from "./database/Storage.js";

// Create database
const areaDatabase = new Storage("myAreas");

// Store data
areaDatabase.set("area_1", {
    name: "My Base",
    owner: "Steve",
    protected: true
});

// Retrieve data
const area = areaDatabase.get("area_1");

// Update atomically
areaDatabase.overwrite("area_1", (current) => ({
    ...current,
    protected: false
}));

// Check existence
if (areaDatabase.has("area_1")) {
    console.log("Area exists!");
}

// Delete
areaDatabase.delete("area_1");

// Statistics
console.log(areaDatabase.getStats());
// { name: "myAreas", size: 100, ... }
```

### 3. Area Protection
```javascript
import { Area, AreaUtils } from "./protection/AreaUtils.js";

// Create area boundary
const areaData = {
    from: { x: 0, y: 0, z: 0 },
    to: { x: 100, y: 100, z: 100 }
};
const area = new Area(areaData);

// Check if block is inside
if (AreaUtils.isInside({ x: 50, y: 50, z: 50 }, area)) {
    console.log("Block is inside protected area!");
}

// Check intersection
const newArea = new Area({ from: { ... }, to: { ... } });
if (AreaUtils.intersects(area, newArea)) {
    console.log("Areas overlap!");
}

// Get area info
const dimensions = AreaUtils.getDimensions(area);
console.log(`${dimensions.width}x${dimensions.depth}x${dimensions.height}`);

const center = AreaUtils.getCenter(area);
console.log(`Center: ${center.print()}`);

// Get all corners
const corners = AreaUtils.getCorners(area);
```
